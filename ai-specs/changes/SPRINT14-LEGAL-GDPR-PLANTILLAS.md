### FASE 3: Implementar Banner de Cookies (Día 3)

#### 3.1 Componente Banner
```typescript
// components/CookieBanner.tsx
'use client';

import { useState, useEffect } from 'react';

export function CookieBanner() {
  const [show, setShow] = useState(false);
  
  useEffect(() => {
    const consent = localStorage.getItem('cookie-consent');
    if (!consent) {
      setShow(true);
    }
  }, []);
  
  const acceptAll = () => {
    localStorage.setItem('cookie-consent', 'all');
    setShow(false);
    // Activar analytics
    enableAnalytics();
  };
  
  const acceptNecessary = () => {
    localStorage.setItem('cookie-consent', 'necessary');
    setShow(false);
  };
  
  if (!show) return null;
  
  return (
    <div className="fixed bottom-0 left-0 right-0 bg-white border-t shadow-lg p-4 z-50">
      <div className="max-w-6xl mx-auto flex items-center justify-between gap-4">
        <div className="flex-1">
          <h3 className="font-bold mb-2">🍪 Usamos cookies</h3>
          <p className="text-sm text-gray-600">
            Utilizamos cookies necesarias para el funcionamiento y opcionales para mejorar tu experiencia.
            {' '}
            <a href="/privacy#cookies" className="underline">Más información</a>
          </p>
        </div>
        
        <div className="flex gap-2">
          <button onClick={acceptNecessary} className="btn-secondary">
            Solo necesarias
          </button>
          <button onClick={acceptAll} className="btn-primary">
            Aceptar todas
          </button>
        </div>
      </div>
    </div>
  );
}
```

// app/api/privacy/download/route.ts
export async function GET(req: Request) {
  const user = await getUser(req);
  
  const data = {
    account: await getUserAccount(user.id),
    queries: await getUserQueries(user.id),
    generated: await getUserGenerated(user.id),
    preferences: await getUserPreferences(user.id),
  };
  
  return Response.json(data, {
    headers: {
      'Content-Disposition': `attachment; filename="opositaia-data-${user.id}.json"`,
    },
  });
}

// app/api/privacy/delete/route.ts
export async function DELETE(req: Request) {
  const user = await getUser(req);
  const { confirmation } = await req.json();
  
  if (confirmation !== 'DELETE_MY_ACCOUNT') {
    return Response.json({ error: 'Confirmation required' }, { status: 400 });
  }
  
  // Eliminar todos los datos
  await deleteUserData(user.id);
  
  return Response.json({ success: true });
}
