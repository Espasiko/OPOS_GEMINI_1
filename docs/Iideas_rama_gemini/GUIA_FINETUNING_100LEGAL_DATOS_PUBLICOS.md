## 3. PREPARAR DATASET COMBINADO

### 3.1 Convertir a JSONL

```python
# legal_dataset_builder.py

import json
import os
from pathlib import Path
from datetime import datetime

class LegalDatasetBuilder:
    """Construye dataset 100% legal"""
    
    def __init__(self, output_file="training_data_legal.jsonl"):
        self.output_file = output_file
        self.records = []
        self.audit_log = {
            "created": datetime.now().isoformat(),
            "components": {},
            "total_records": 0
        }
    
    def add_laws_from_boe(self, laws_dir="data/laws"):
        """Agregar leyes BOE (70% del dataset)"""
        print(f"📜 Procesando leyes BOE desde {laws_dir}...")
        
        for law_file in Path(laws_dir).glob("*.txt"):
            with open(law_file, 'r', encoding='utf-8') as f:
                content = f.read()
            
            # Dividir en párrafos significativos
            paragraphs = content.split('\n\n')
            
            for i, para in enumerate(paragraphs):
                if len(para.strip()) > 100:
                    self.records.append({
                        "instruction": "Explica este concepto legal basado en leyes españolas",
                        "input": "",
                        "output": para.strip(),
                        "source": f"BOE_{law_file.stem}_{i}",
                        "component": "public_laws",
                        "legal_status": "public_domain",
                        "attribution": "Gobierno de España - BOE"
                    })
        
        self.audit_log["components"]["public_laws"] = len([r for r in self.records if r["component"] == "public_laws"])
        print(f"  ✅ Agregadas {self.audit_log['components']['public_laws']} secciones legales")
    
    def add_jurisprudence(self, sentences_file="data/jurisprudence/sentences.json"):
        """Agregar jurisprudencia pública (10% del dataset)"""
        print(f"📋 Procesando jurisprudencia desde {sentences_file}...")
        
        if not Path(sentences_file).exists():
            print(f"  ⚠️ Archivo no encontrado: {sentences_file}")
            return
        
        with open(sentences_file, 'r', encoding='utf-8') as f:
            sentences = json.load(f)
        
        for sentence in sentences[:10000]:  # Máximo 10K
            self.records.append({
                "instruction": "Analiza esta sentencia judicial",
                "input": f"Tribunal: {sentence.get('tribunal', 'N/A')}",
                "output": sentence.get('texto', ''),
                "source": f"SENTENCE_{sentence.get('id', 'unknown')}",
                "component": "jurisprudence",
                "legal_status": "public_access",
                "attribution": "Poder Judicial de España"
            })
        
        self.audit_log["components"]["jurisprudence"] = len([r for r in self.records if r["component"] == "jurisprudence"])
        print(f"  ✅ Agregadas {self.audit_log['components']['jurisprudence']} sentencias")
    
    def add_official_tests(self, tests_dir="data/tests"):
        """Agregar tests oficiales (5% del dataset)"""
        print(f"📝 Procesando tests oficiales desde {tests_dir}...")
        
        for test_file in Path(tests_dir).glob("*.json"):
            with open(test_file, 'r', encoding='utf-8') as f:
                tests = json.load(f)
            
            for test in tests:
                self.records.append({
                    "instruction": test.get('pregunta', ''),
                    "input": "",
                    "output": test.get('respuesta', ''),
                    "source": f"TEST_{test_file.stem}",
                    "component": "official_tests",
                    "legal_status": "public_domain",
                    "attribution": f"Ministerio - {test_file.stem}"
                })
        
        self.audit_log["components"]["official_tests"] = len([r for r in self.records if r["component"] == "official_tests"])
        print(f"  ✅ Agregadas {self.audit_log['components']['official_tests']} preguntas de tests")
    
    def add_personal_simulacros(self, simulacros_file="data/personal/simulacros.jsonl"):
        """Agregar tus propios simulacros (15% del dataset)"""
        print(f"🎓 Procesando tus simulacros desde {simulacros_file}...")
        
        if not Path(simulacros_file).exists():
            print(f"  ⚠️ Archivo no encontrado: {simulacros_file}")
            return
        
        with open(simulacros_file, 'r', encoding='utf-8') as f:
            for i, line in enumerate(f):
                record = json.loads(line)
                record["component"] = "personal_simulacros"
                record["legal_status"] = "personal_copyright"
                record["attribution"] = "Tuyo"
                self.records.append(record)
        
        self.audit_log["components"]["personal_simulacros"] = len([r for r in self.records if r["component"] == "personal_simulacros"])
        print(f"  ✅ Agregados {self.audit_log['components']['personal_simulacros']} simulacros propios")
    
    def validate_composition(self):
        """Verificar que la composición es legal"""
        print("\n🔍 Validando composición del dataset...")
        
        composition = {
            "public_laws": self.audit_log["components"].get("public_laws", 0),
            "jurisprudence": self.audit_log["components"].get("jurisprudence", 0),
            "official_tests": self.audit_log["components"].get("official_tests", 0),
            "personal_simulacros": self.audit_log["components"].get("personal_simulacros", 0),
        }
        
        total = sum(composition.values())
        percentages = {k: (v/total*100 if total > 0 else 0) for k, v in composition.items()}
        
        print(f"\n  Composición:")
        print(f"  ├─ Leyes públicas: {composition['public_laws']} ({percentages['public_laws']:.0f}%)")
        print(f"  ├─ Jurisprudencia: {composition['jurisprudence']} ({percentages['jurisprudence']:.0f}%)")
        print(f"  ├─ Tests oficiales: {composition['official_tests']} ({percentages['official_tests']:.0f}%)")
        print(f"  └─ Simulacros propios: {composition['personal_simulacros']} ({percentages['personal_simulacros']:.0f}%)")
        
        # Verificar que todo sea legal
        all_legal = all(
            r["legal_status"] in ["public_domain", "public_access", "personal_copyright"]
            for r in self.records
        )
        
        if all_legal:
            print(f"\n  ✅ DATASET 100% LEGAL")
        else:
            print(f"\n  ❌ ADVERTENCIA: Algunos registros no son legales")
        
        return all_legal
    
    def save_dataset(self):
        """Guardar dataset en JSONL"""
        print(f"\n💾 Guardando dataset en {self.output_file}...")
        
        with open(self.output_file, 'w', encoding='utf-8') as f:
            for record in self.records:
                f.write(json.dumps(record, ensure_ascii=False) + '\n')
        
        self.audit_log["total_records"] = len(self.records)
        
        # Guardar audit log
        with open("DATASET_AUDIT.json", 'w', encoding='utf-8') as f:
            json.dump(self.audit_log, f, indent=2, ensure_ascii=False)
        
        print(f"  ✅ {len(self.records)} registros guardados")
        print(f"  ✅ Audit log guardado en DATASET_AUDIT.json")

# USO
if __name__ == "__main__":
    builder = LegalDatasetBuilder()
    
    # Agregar componentes
    builder.add_laws_from_boe("data/laws")
    builder.add_jurisprudence("data/jurisprudence/sentences.json")
    builder.add_official_tests("data/tests")
    builder.add_personal_simulacros("data/personal/simulacros.jsonl")
    
    # Validar
    builder.validate_composition()
    
    # Guardar
    builder.save_dataset()
    
    print("\n" + "="*50)
    print("✅ DATASET LEGAL 100% CREADO Y VALIDADO")
    print("="*50)
```
